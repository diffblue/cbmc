40static bool
41depends_on(const exprt &expr, std::unordered_set<irep_idt> identifiers)
42{
43 const std::unordered_set<irep_idt> ids = find_symbol_identifiers(expr);
44 for(const auto &id : ids)

58std::unordered_set<irep_idt> gen_loop_locals_set(
59 goto_functiont &goto_function,
60 const dfcc_loop_nesting_graph_nodet &loop_node)
61{
62 std::unordered_set<irep_idt> loop_locals;
63
64 const auto &loop = loop_node.instructions;
65
66 Forall_goto_program_instructions(i_it, goto_function.body)
67 {
68 // Only variables declared inside the loop are loop locals.
69 if(i_it->is_decl() && loop.contains(i_it))
70 {
71 loop_locals.insert(i_it->decl_symbol().identifier());
72 }
73 }
74
75 return loop_locals;
76}

79static std::unordered_set<irep_idt>
80find_symbol_identifiers(const goto_programt &src)
81{
82 std::unordered_set<irep_idt> identifiers;
83 for(const auto &instruction : src.instructions)
84 {
85 // compute forward edges first
86 switch(instruction.type())
87 {
88 case ASSERT:
89 case ASSUME:
90 case GOTO:
91 find_symbols(instruction.condition(), identifiers);
92 break;
93
94 case FUNCTION_CALL:
95 find_symbols(instruction.call_lhs(), identifiers);
96 for(const auto &e : instruction.call_arguments())
97 find_symbols(e, identifiers);
98 break;
99 case ASSIGN:
100 case OTHER:
101
102 case SET_RETURN_VALUE:
103 case DECL:
104 case DEAD:
105 for(const auto &e : instruction.code().operands())
106 {
107 find_symbols(e, identifiers);
108 }
109 break;
110
111 case END_THREAD:
112 case END_FUNCTION:
113 case ATOMIC_BEGIN:
114 case ATOMIC_END:
115 case SKIP:
116 case LOCATION:
117 case CATCH:
118 case THROW:
119 case START_THREAD:
120 break;
121 case NO_INSTRUCTION_TYPE:
122 case INCOMPLETE_GOTO:
123 UNREACHABLE;
124 break;
125 }
126 }
127 return identifiers;
128}

134static assignst dfcc_infer_loop_assigns_for_loop(
135 const local_may_aliast &local_may_alias,
136 goto_functiont &goto_function,
137 const dfcc_loop_nesting_graph_nodet &loop,
138 const std::unordered_set<irep_idt> &candidate_targets,
139 message_handlert &message_handler,
140 const namespacet &ns)
141{
142 // infer
143 assignst assigns;
144 infer_loop_assigns(local_may_alias, loop.instructions, assigns);
145
146 // compute locals
147 std::unordered_set<irep_idt> loop_locals =
148 gen_loop_locals_set(goto_function, loop);
149
150 // widen or drop targets that depend on loop-locals or are non-constant,
151 // ie. depend on other locations assigned by the loop.
152 // e.g: if the loop assigns {i, a[i]}, then a[i] is non-constant.
153 havoc_utils_can_forward_propagatet is_constant(assigns, ns);
154 assignst result;
155 for(const auto &expr : assigns)
156 {
157 // Skip targets that only depend on non-visible identifiers.
158 if(!depends_on(expr, candidate_targets))
159 {
160 continue;
161 }
162
163 if(depends_on(expr, loop_locals))
164 {
165 // Target depends on loop locals, attempt widening to the root object
166 auto root_objects = dfcc_root_objects(expr);
167 for(const auto &root_object : root_objects)
168 {
169 if(!depends_on(root_object, loop_locals))
170 {
171 address_of_exprt address_of_root_object(root_object);
172 address_of_root_object.add_source_location() =
173 root_object.source_location();
174 result.emplace(
175 make_object_whole_call_expr(address_of_root_object, ns));
176 }
177 }
178 }
179 else
180 {
181 address_of_exprt address_of_expr(expr);
182 address_of_expr.add_source_location() = expr.source_location();
183 // Widen assigns targets to object_whole if `expr` is a dereference or
184 // with constant address.
185 if(expr.id() == ID_dereference || !is_constant(address_of_expr))
186 {
187 // Target address is not constant, widening to the whole object
188 result.emplace(make_object_whole_call_expr(address_of_expr, ns));
189 }
190 else
191 {
192 result.emplace(expr);
193 }
194 }
195 }
196
197 return result;
198}

218struct originst
219{
220 std::set<irep_idt> globals;
221 std::set<irep_idt> locals;
222 std::set<std::size_t> parameters;
223 bool unknown = false;
224
226 bool merge(const originst &other)
227 {
228 const std::size_t size = globals.size() + locals.size() + parameters.size();
229 const bool was_unknown = unknown;
230 globals.insert(other.globals.begin(), other.globals.end());
231 locals.insert(other.locals.begin(), other.locals.end());
232 parameters.insert(other.parameters.begin(), other.parameters.end());
233 unknown |= other.unknown;
234 return was_unknown != unknown ||
235 size != globals.size() + locals.size() + parameters.size();
236 }
237};

273 void compute(const std::vector<irep_idt> &roots)
274 {
275 std::vector<irep_idt> work = roots;
276 std::set<irep_idt> seen(roots.begin(), roots.end());
277 while(!work.empty())
278 {
279 const irep_idt id = work.back();
280 work.pop_back();
281 functions.push_back(id);
282 for(const auto &instruction : body_of(id).instructions)
283 {
284 if(
285 instruction.is_function_call() &&
286 instruction.call_function().id() == ID_symbol)
287 {
288 const irep_idt &callee =
289 to_symbol_expr(instruction.call_function()).identifier();
290 if(has_body(callee) && seen.insert(callee).second)
291 work.push_back(callee);
292 }
293 }
294 }
295 // Iterate to a fixpoint for recursive functions: summaries only grow.
296 bool changed = true;
297 while(changed)
298 {
299 changed = false;
300 for(const auto &id : functions)
301 {
302 write_summaryt summary = analyze(id);
303 if(!(summary == summaries[id]))
304 {
305 summaries[id] = std::move(summary);
306 changed = true;
307 }
308 }
309 }
310 }
311
312 const write_summaryt &operator[](const irep_idt &id)
313 {
314 return summaries[id];
315 }

317 bool has_pointers(const typet &type)
318 {
319 auto entry = pointer_types.find(type);
320 if(entry != pointer_types.end())
321 return entry->second;
322 // Assume none while the type is visited, for recursive types.
323 pointer_types[type] = false;
324 bool result = false;
325 if(type.id() == ID_pointer)
326 result = true;
327 else if(
328 type.id() == ID_struct_tag || type.id() == ID_union_tag ||
329 type.id() == ID_c_enum_tag)
330 {
331 result = has_pointers(follow_tags(type, ns));
332 }
333 else if(type.id() == ID_struct || type.id() == ID_union)
334 {
335 for(const auto &component : to_struct_union_type(type).components())
336 result = result || has_pointers(component.type());
337 }
338 else if(type.id() == ID_array || type.id() == ID_vector)
339 result = has_pointers(to_type_with_subtype(type).subtype());
340 pointer_types[type] = result;
341 return result;
342 }

375 write_summaryt analyze(const irep_idt &id)
376 {
377 const goto_functiont &function = goto_functions.function_map.at(id);
378 variables.clear();
379 parameter_index.clear();
380 current = write_summaryt();
381 for(std::size_t i = 0; i < function.parameter_identifiers.size(); ++i)
382 {
383 const irep_idt &parameter = function.parameter_identifiers[i];
384 parameter_index[parameter] = i;
385 variables[parameter].parameters.insert(i);
386 }
387 do
388 {
389 variables_changed = false;
390 for(const auto &instruction : function.body.instructions)
391 analyze(instruction);
392 } while(variables_changed);
393 return current;
394 }

396 void analyze(const goto_programt::instructiont &instruction)
397 {
398 if(instruction.is_assign())
399 write(instruction.assign_lhs(), origins(instruction.assign_rhs()));
400 else if(instruction.is_set_return_value())
401 current.result.merge(origins(instruction.return_value()));
402 else if(instruction.is_function_call())
403 call(instruction);
404 else if(instruction.is_other())
405 {
406 const codet &code = instruction.get_other();
407 const irep_idt &statement = code.get_statement();
408 if(
409 statement == ID_array_set || statement == ID_array_copy ||
410 statement == ID_array_replace || statement == ID_havoc_object)
411 {
412 originst content;
413 content.unknown = true;
414 write_through(origins(code.op0()), content);
415 }
416 }
417 }

419 void call(const goto_programt::instructiont &instruction)
420 {
421 const exprt &function = instruction.call_function();
422 const exprt &lhs = instruction.call_lhs();
423 originst unknown;
424 unknown.unknown = true;
425 if(function.id() != ID_symbol)
426 {
427 current.unknown = true;
428 if(lhs.is_not_nil())
429 write(lhs, unknown);
430 return;
431 }
432 const irep_idt &callee = to_symbol_expr(function).identifier();
433 if(!has_body(callee))
434 {
435 // Built-ins write nothing but their result; any other function without
436 // a body may write anything.
437 if(!dfcc_is_cprover_function_symbol(callee))
438 current.unknown = true;
439 if(lhs.is_not_nil())
440 write(lhs, unknown);
441 return;
442 }
443 const write_summaryt summary = summaries[callee];
444 const auto &arguments = instruction.call_arguments();
445 current.globals.insert(summary.globals.begin(), summary.globals.end());
446 current.unknown |= summary.unknown;
447 for(const auto i : summary.parameters)
448 {
449 if(i < arguments.size())
450 write_through(origins(arguments[i]), unknown);
451 else
452 current.unknown = true;
453 }
454 if(lhs.is_not_nil())
455 {
456 originst result;
457 result.globals = summary.result.globals;
458 result.unknown = summary.result.unknown || !summary.result.locals.empty();
459 for(const auto i : summary.result.parameters)
460 {
461 if(i < arguments.size())
462 result.merge(origins(arguments[i]));
463 else
464 result.unknown = true;
465 }
466 write(lhs, result);
467 }
468 }

471 void write_through(const originst &target, const originst &content)
472 {
473 current.globals.insert(target.globals.begin(), target.globals.end());
474 current.parameters.insert(
475 target.parameters.begin(), target.parameters.end());
476 current.unknown |= target.unknown;
477 for(const auto &local : target.locals)
478 variables_changed |= variables[local].merge(content);
479 }

482 void write(const exprt &lhs, const originst &content)
483 {
484 if(lhs.id() == ID_symbol)
485 {
486 const irep_idt &id = to_symbol_expr(lhs).identifier();
487 if(is_global(id))
488 current.globals.insert(id);
489 else
490 variables_changed |= variables[id].merge(content);
491 }
492 else if(
493 lhs.id() == ID_member || lhs.id() == ID_index ||
494 lhs.id() == ID_byte_extract_little_endian ||
495 lhs.id() == ID_byte_extract_big_endian || lhs.id() == ID_typecast)
496 {
497 write(lhs.operands().front(), content);
498 }
499 else if(lhs.id() == ID_dereference)
500 write_through(origins(to_dereference_expr(lhs).pointer()), content);
501 else if(lhs.id() == ID_if)
502 {
503 write(to_if_expr(lhs).true_case(), content);
504 write(to_if_expr(lhs).false_case(), content);
505 }
506 else
507 current.unknown = true;
508 }
509
511 originst object_origins(const exprt &object)
512 {
513 originst result;
514 if(object.id() == ID_symbol)
515 {
516 const irep_idt &id = to_symbol_expr(object).identifier();
517 if(is_global(id))
518 result.globals.insert(id);
519 else
520 result.locals.insert(id);
521 }
522 else if(
523 object.id() == ID_member || object.id() == ID_index ||
524 object.id() == ID_byte_extract_little_endian ||
525 object.id() == ID_byte_extract_big_endian)
526 {
527 return object_origins(object.operands().front());
528 }
529 else if(object.id() == ID_dereference)
530 return origins(to_dereference_expr(object).pointer());
531 else if(object.id() == ID_string_constant || object.id() == ID_constant)
532 {
533 }
534 else
535 result.unknown = true;
536 return result;
537 }
538
540 originst origins(const exprt &expr)
541 {
542 originst result;
543 if(expr.id() == ID_address_of)
544 return object_origins(to_address_of_expr(expr).object());
545 if(!has_pointers(expr.type()) && expr.id() != ID_typecast)
546 return result;
547 if(expr.id() == ID_symbol)
548 {
549 const irep_idt &id = to_symbol_expr(expr).identifier();
550 if(is_global(id))
551 result.unknown = true;
552 else
553 result = variables[id];
554 }
555 else if(expr.id() == ID_dereference)
556 {
557 // A value loaded from memory: from the analysis' locals, or unknown.
558 const originst target = origins(to_dereference_expr(expr).pointer());
559 result.unknown =
560 target.unknown || !target.globals.empty() || !target.parameters.empty();
561 for(const auto &local : target.locals)
562 result.merge(variables[local]);
563 }
564 else if(expr.id() == ID_typecast)
565 {
566 const exprt &op = to_typecast_expr(expr).op();
567 if(!has_pointers(expr.type()))
568 return result;
569 // A pointer made from an integer may point anywhere.
570 if(!has_pointers(op.type()))
571 result.unknown = true;
572 else
573 result = origins(op);
574 }
575 else if(expr.id() == ID_constant || expr.id() == ID_nil)
576 {
577 }
578 else if(expr.id() == ID_if)
579 {
580 result = origins(to_if_expr(expr).true_case());
581 result.merge(origins(to_if_expr(expr).false_case()));
582 }
583 else if(expr.id() == ID_side_effect || expr.id() == ID_nondet_symbol)
584 result.unknown = true;
585 else
586 {
587 // Member and index of a value, pointer arithmetic, `with`, struct
588 // literals...: the pointers in the result come from the operands.
589 for(const auto &op : expr.operands())
590 result.merge(origins(op));
591 }
592 return result;
593 }

609class write_summary_bodiest
610{
611public:
612 write_summary_bodiest(goto_functionst &goto_functions, const namespacet &ns)
613 : goto_functions(goto_functions)
614 {
615 std::vector<irep_idt> roots;
616 for(const auto &entry : goto_functions.function_map)
617 {
618 if(
619 dfcc_is_cprover_function_symbol(entry.first) &&
620 entry.second.body_available())
621 {
622 roots.push_back(entry.first);
623 }
624 }
625 write_summariest summaries(goto_functions, ns);
626 summaries.compute(roots);
627 for(const auto &id : roots)
628 {
629 goto_functiont &function = goto_functions.function_map.at(id);
630 goto_programt body;
631 if(make_body(id, function, summaries[id], summaries, ns, body))
632 {
633 saved.emplace_back(id, std::move(function.body));
634 function.body = std::move(body);
635 }
636 }
637 }
638
639 ~write_summary_bodiest()
640 {
641 for(auto &entry : saved)
642 goto_functions.function_map.at(entry.first).body =
643 std::move(entry.second);
644 }

653 static bool write_through(
654 const exprt &value,
655 write_summariest &summaries,
656 const namespacet &ns,
657 const source_locationt &location,
658 goto_programt &body)
659 {
660 const typet &type = value.type();
661 if(type.id() == ID_pointer)
662 {
663 const typet &pointee = to_pointer_type(type).base_type();
664 const bool sized = pointee.id() != ID_empty && pointee.id() != ID_code &&
665 !(pointee.id() == ID_struct_tag &&
666 follow_tags(pointee, ns).get_bool(ID_incomplete));
667 const exprt pointer =
668 sized ? value : exprt(typecast_exprt(value, pointer_type(char_type())));
669 const dereference_exprt target(pointer);
670 body.add(goto_programt::make_assignment(
671 target, side_effect_expr_nondett(target.type(), location), location));
672 return true;
673 }
674 if(!summaries.has_pointers(type))
675 return true;
676 if(type.id() == ID_struct_tag || type.id() == ID_struct)
677 {
678 for(const auto &component :
679 to_struct_type(follow_tags(type, ns)).components())
680 {
681 if(!write_through(
682 member_exprt(value, component), summaries, ns, location, body))
683 {
684 return false;
685 }
686 }
687 return true;
688 }
689 return false;
690 }
691
693 static bool make_body(
694 const irep_idt &id,
695 const goto_functiont &function,
696 const write_summaryt &summary,
697 write_summariest &summaries,
698 const namespacet &ns,
699 goto_programt &body)
700 {
701 if(summary.unknown)
702 return false;
703 const symbolt &symbol = ns.lookup(id);
704 const source_locationt &location = symbol.location;
705 for(const auto &global : summary.globals)
706 {
707 const symbol_exprt target = ns.lookup(global).symbol_expr();
708 body.add(goto_programt::make_assignment(
709 target, side_effect_expr_nondett(target.type(), location), location));
710 }
711 for(const auto i : summary.parameters)
712 {
713 if(i >= function.parameter_identifiers.size())
714 return false;
715 const symbol_exprt parameter =
716 ns.lookup(function.parameter_identifiers[i]).symbol_expr();
717 if(!write_through(parameter, summaries, ns, location, body))
718 return false;
719 }
720 const typet &return_type = to_code_type(symbol.type).return_type();
721 if(return_type.id() != ID_empty)
722 {
723 exprt result = side_effect_expr_nondett(return_type, location);
724 if(return_type.id() == ID_pointer && !summary.result.unknown)
725 {
726 // Choose among the objects the result may point into.
727 std::vector<exprt> candidates;
728 for(const auto &global : summary.result.globals)
729 {
730 candidates.push_back(typecast_exprt::conditional_cast(
731 address_of_exprt(ns.lookup(global).symbol_expr()), return_type));
732 }
733 for(const auto i : summary.result.parameters)
734 {
735 if(i >= function.parameter_identifiers.size())
736 return false;
737 const symbol_exprt parameter =
738 ns.lookup(function.parameter_identifiers[i]).symbol_expr();
739 if(parameter.type().id() != ID_pointer)
740 return false;
741 candidates.push_back(
742 typecast_exprt::conditional_cast(parameter, return_type));
743 }
744 for(const auto &candidate : candidates)
745 {
746 result = if_exprt(
747 side_effect_expr_nondett(bool_typet(), location),
748 candidate,
749 result);
750 }
751 }
752 else if(summaries.has_pointers(return_type) || summary.result.unknown)
753 {
754 // A result whose pointers the summary cannot name.
755 if(summaries.has_pointers(return_type))
756 return false;
757 }
758 body.add(goto_programt::make_set_return_value(result, location));
759 }
760 body.add(goto_programt::make_end_function(location));
761 return true;
762 }

786void dfcc_infer_loop_assigns_for_function(
787 std::map<std::size_t, assignst> &inferred_loop_assigns_map,
788 goto_functionst &goto_functions,
789 const goto_functiont &goto_function,
790 message_handlert &message_handler,
791 const namespacet &ns)
792{
793 messaget log(message_handler);
794
795 // Collect all candidate targets---identifiers visible in `goto_function`.
796 const auto candidate_targets = find_symbol_identifiers(goto_function.body);
797
798 // We infer loop assigns based on the copy of `goto_function`.
799 goto_functiont goto_function_copy;
800 goto_function_copy.copy_from(goto_function);
801
802 // Build the loop id map before inlining attempt. So that we can later
803 // distinguish loops in the original body and loops added by inlining.
804 const auto loop_nesting_graph =
805 build_loop_nesting_graph(goto_function_copy.body);
806 auto topsorted = loop_nesting_graph.topsort();
807 // skip function without loop.
808 if(topsorted.empty())
809 return;
810
811 // Map from targett in `goto_function_copy` to loop number.
812 std::
813 unordered_map<goto_programt::const_targett, std::size_t, const_target_hash>
814 loop_number_map;
815
816 for(const auto id : topsorted)
817 {
818 loop_number_map.emplace(
819 loop_nesting_graph[id].head, loop_nesting_graph[id].latch->loop_number);
820 }
821
822 // We avoid inlining `malloc` and `free` whose variables are not assigns.
823 auto malloc_body = goto_functions.function_map.extract(irep_idt("malloc"));
824 auto free_body = goto_functions.function_map.extract(irep_idt("free"));
825
826 // Replace the bodies of the functions that DFCC does not instrument by
827 // summaries of their writes, see `write_summary_bodiest`.
828 write_summary_bodiest summary_bodies(goto_functions, ns);
829
830 // Inline all function calls in goto_function_copy; this is best-effort
831 // inlining, we can safely ignore warnings here.
832 null_message_handlert null_message_handler;
833 goto_program_inline(
834 goto_functions, goto_function_copy.body, ns, null_message_handler);
835 // Update the body to make sure all goto correctly jump to valid targets.
836 goto_function_copy.body.update();
837 // Build the loop graph after inlining.
838 const auto inlined_loop_nesting_graph =
839 build_loop_nesting_graph(goto_function_copy.body);
840
841 // Alias analysis.
842 remove_dead_object_assignment(goto_function_copy);
843 local_may_aliast local_may_alias(goto_function_copy);
844
845 auto inlined_topsorted = inlined_loop_nesting_graph.topsort();
846
847 for(const auto inlined_id : inlined_topsorted)
848 {
849 // We only infer loop assigns for loops in the original function.
850 if(
851 loop_number_map.find(inlined_loop_nesting_graph[inlined_id].head) !=
852 loop_number_map.end())
853 {
854 const auto loop_number =
855 loop_number_map[inlined_loop_nesting_graph[inlined_id].head];
856 const auto inlined_loop = inlined_loop_nesting_graph[inlined_id];
857
858 inferred_loop_assigns_map[loop_number] = dfcc_infer_loop_assigns_for_loop(
859 local_may_alias,
860 goto_function_copy,
861 inlined_loop,
862 candidate_targets,
863 message_handler,
864 ns);
865 }
866 }
867 // Restore the function boyd of `malloc` and `free`.
868 goto_functions.function_map.insert(std::move(malloc_body));
869 goto_functions.function_map.insert(std::move(free_body));
870}
