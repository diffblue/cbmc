375// The value of the digit ch in bases up to 36, and 36 for any other character
376#define __CPROVER_STRTO_DIGIT_VALUE(ch) \
377 ((ch) >= '0' && (ch) <= '9' ? (ch) - '0' \
378 : (ch) >= 'a' && (ch) <= 'z' ? (ch) - 'a' + 10 \
379 : (ch) >= 'A' && (ch) <= 'Z' ? (ch) - 'A' + 10 \
380 : 36)
381
394unsigned long long __CPROVER_strto_scan(
395 const char *nptr,
396 char **endptr,
397 int base,
398 _Bool c23,
399 _Bool *negative,
400 _Bool *overflow)
401{
402__CPROVER_HIDE:;
403#ifdef __CPROVER_STRING_ABSTRACTION
404 __CPROVER_precondition(
405 __CPROVER_is_zero_string(nptr), "zero-termination of argument of strto*");
406#endif
407
408 *negative = 0;
409 *overflow = 0;
410
411 if(base == 1 || base < 0 || base > 36)
412 {
413 errno = EINVAL;
414 return 0;
415 }
416
417 __CPROVER_size_t i = 0;
418 while(isspace((unsigned char)nptr[i]))
419 ++i;
420
421 _Bool is_negative = 0;
422 if(nptr[i] == '-')
423 {
424 is_negative = 1;
425 ++i;
426 }
427 else if(nptr[i] == '+')
428 ++i;
429
430 // A prefix is only part of the subject sequence if a digit follows it.
431 if(
432 (base == 0 || base == 16) && nptr[i] == '0' &&
433 (nptr[i + 1] == 'x' || nptr[i + 1] == 'X') &&
434 __CPROVER_STRTO_DIGIT_VALUE(nptr[i + 2]) < 16)
435 {
436 base = 16;
437 i += 2;
438 }
439 else if(
440 c23 && (base == 0 || base == 2) && nptr[i] == '0' &&
441 (nptr[i + 1] == 'b' || nptr[i + 1] == 'B') &&
442 __CPROVER_STRTO_DIGIT_VALUE(nptr[i + 2]) < 2)
443 {
444 base = 2;
445 i += 2;
446 }
447 else if(base == 0)
448 base = nptr[i] == '0' ? 8 : 10;
449
450 const __CPROVER_size_t first_digit = i;
451 unsigned long long result = 0;
452 _Bool has_overflow = 0;
453 for(;; ++i)
454 {
455 int digit = __CPROVER_STRTO_DIGIT_VALUE(nptr[i]);
456 if(digit >= base)
457 break;
458 // keep consuming digits after an overflow, as endptr must point past the
459 // subject sequence
460 if(
461 !has_overflow &&
462 (__builtin_mul_overflow(result, (unsigned long long)base, &result) ||
463 __builtin_add_overflow(result, (unsigned long long)digit, &result)))
464 {
465 has_overflow = 1;
466 }
467 }
468
469 if(i == first_digit)
470 {
471 // no conversion performed
472 if(endptr != 0)
473 *endptr = (char *)nptr;
474 return 0;
475 }
476
477 if(endptr != 0)
478 *endptr = (char *)nptr + i;
479
480 *negative = is_negative;
481 *overflow = has_overflow;
482 return has_overflow ? ULLONG_MAX : result;
483}
