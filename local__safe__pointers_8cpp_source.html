38static std::optional<goto_null_checkt> get_null_checked_expr(const exprt &expr)
39{
40 exprt normalized_expr = expr;
41 // If true, then a null check is made when test `expr` passes; if false,
42 // one is made when it fails.
43 bool checked_when_taken = true;
44
45 // Reduce some roundabout ways of saying "x != null", e.g. "!(x == null)".
46 while(normalized_expr.id() == ID_not)
47 {
48 normalized_expr = to_not_expr(normalized_expr).op();
49 checked_when_taken = !checked_when_taken;
50 }
51
52 if(normalized_expr.id() == ID_equal)
53 {
54 normalized_expr.id(ID_notequal);
55 checked_when_taken = !checked_when_taken;
56 }
57
58 if(normalized_expr.id() == ID_notequal)
59 {
60 const exprt &op0 = skip_typecast(to_notequal_expr(normalized_expr).op0());
61 const exprt &op1 = skip_typecast(to_notequal_expr(normalized_expr).op1());
62
63 if(op0.type().id() == ID_pointer &&
64 op0 == null_pointer_exprt(to_pointer_type(op0.type())))
65 {
66 return { { checked_when_taken, op1 } };
67 }
68
69 if(op1.type().id() == ID_pointer &&
70 op1 == null_pointer_exprt(to_pointer_type(op1.type())))
71 {
72 return { { checked_when_taken, op0 } };
73 }
74 }
75
76 return {};
77}

85static void
86collect_if_guarded_dereferences(const exprt &expr, std::vector<exprt> &result)
87{
88 if(const auto *if_expr = expr_try_dynamic_cast<if_exprt>(expr))
89 {
90 if(auto check = get_null_checked_expr(if_expr->cond()))
91 {
92 // The guarded operand is the one evaluated when the pointer is non-null.
93 const exprt &guarded_operand = check->checked_when_taken
94 ? if_expr->true_case()
95 : if_expr->false_case();
96 const exprt checked_pointer = skip_typecast(check->checked_expr);
97 for(auto it = guarded_operand.depth_begin(),
98 end = guarded_operand.depth_end();
99 it != end;
100 ++it)
101 {
102 if(
103 it->id() == ID_dereference &&
104 skip_typecast(to_dereference_expr(*it).pointer()) == checked_pointer)
105 {
106 result.push_back(*it);
107 }
108 }
109 }
110 }
111
112 for(const auto &op : expr.operands())
113 collect_if_guarded_dereferences(op, result);
114}

121void local_safe_pointerst::operator()(const goto_programt &goto_program)
122{
123 std::set<exprt, type_comparet> checked_expressions(type_comparet{});
124
125 for(const auto &instruction : goto_program.instructions)
126 {
127 // Record dereferences guarded by a conditional null-check within this
128 // instruction (e.g. `(p != null) ? *p : x`, as produced by if-conversion).
129 // This is independent of the control-flow tracking below.
130 {
131 std::vector<exprt> guarded;
132 instruction.apply([&guarded](const exprt &e)
133 { collect_if_guarded_dereferences(e, guarded); });
134 if(!guarded.empty())
135 {
136 auto &guarded_set =
137 if_guarded_dereferences[instruction.location_number];
138 for(auto &deref : guarded)
139 guarded_set.insert(std::move(deref));
140 }
141 }
142
143 // Handle control-flow convergence pessimistically:
144 if(instruction.incoming_edges.size() > 1)
145 checked_expressions.clear();
146 // Retrieve working set for forward GOTOs:
147 else if(instruction.is_target())
148 {
149 auto findit = non_null_expressions.find(instruction.location_number);
150 if(findit != non_null_expressions.end())
151 checked_expressions = findit->second;
152 else
153 {
154 checked_expressions = std::set<exprt, type_comparet>(type_comparet{});
155 }
156 }
157
158 // Save the working set at this program point:
159 if(!checked_expressions.empty())
160 {
161 non_null_expressions.emplace(
162 instruction.location_number, checked_expressions);
163 }
164
165 switch(instruction.type())
166 {
167 // Instruction types that definitely don't write anything, and therefore
168 // can't store a null pointer:
169 case DECL:
170 case DEAD:
171 case ASSERT:
172 case SKIP:
173 case LOCATION:
174 case SET_RETURN_VALUE:
175 case THROW:
176 case CATCH:
177 case END_FUNCTION:
178 case ATOMIC_BEGIN:
179 case ATOMIC_END:
180 break;
181
182 // Possible checks:
183 case ASSUME:
184 if(auto assume_check = get_null_checked_expr(instruction.condition()))
185 {
186 if(assume_check->checked_when_taken)
187 checked_expressions.insert(assume_check->checked_expr);
188 }
189
190 break;
191
192 case GOTO:
193 if(!instruction.is_backwards_goto())
194 {
195 // Copy current state to GOTO target:
196
197 auto target_emplace_result =
198 non_null_expressions.emplace(
199 instruction.get_target()->location_number, checked_expressions);
200
201 // If the target already has a state entry then it is a control-flow
202 // merge point and everything will be assumed maybe-null in any case.
203 if(target_emplace_result.second)
204 {
205 if(
206 auto conditional_check =
207 get_null_checked_expr(instruction.condition()))
208 {
209 // Add the GOTO condition to either the target or current state,
210 // as appropriate:
211 if(conditional_check->checked_when_taken)
212 {
213 target_emplace_result.first->second.insert(
214 conditional_check->checked_expr);
215 }
216 else
217 checked_expressions.insert(conditional_check->checked_expr);
218 }
219 }
220 }
221
222 break;
223
224 case ASSIGN:
225 case START_THREAD:
226 case END_THREAD:
227 case FUNCTION_CALL:
228 case OTHER:
229 case INCOMPLETE_GOTO:
230 case NO_INSTRUCTION_TYPE:
231 // Pessimistically assume all other instructions might overwrite any
232 // pointer with a possibly-null value.
233 checked_expressions.clear();
234 break;
235 }
236 }
237}

244void local_safe_pointerst::output(
245 std::ostream &out,
246 const goto_programt &goto_program,
247 const namespacet &ns)
248{
249 for(const auto &instruction : goto_program.instructions)
250 {
251 out << "**** " << instruction.location_number << " "
252 << instruction.source_location() << "\n";
253
254 out << "Non-null expressions: ";
255
256 auto findit = non_null_expressions.find(instruction.location_number);
257 if(findit == non_null_expressions.end())
258 out << "{}";
259 else
260 {
261 out << "{";
262 bool first = true;
263 for(const exprt &expr : findit->second)
264 {
265 if(!first)
266 out << ", ";
267 first = true;
268 format_rec(out, expr);
269 }
270 out << "}";
271 }
272
273 out << '\n';
274 instruction.output(out);
275 out << '\n';
276 }

288void local_safe_pointerst::output_safe_dereferences(
289 std::ostream &out,
290 const goto_programt &goto_program,

292{
293 for(const auto &instruction : goto_program.instructions)
294 {
295 out << "**** " << instruction.location_number << " "

297
298 out << "Safe (known-not-null) dereferences: ";
299
300 auto findit = non_null_expressions.find(instruction.location_number);
301 if(findit == non_null_expressions.end())
302 out << "{}";
303 else
304 {
305 out << "{";
306 bool first = true;
307 instruction.apply([&first, &out](const exprt &e) {
308 for(auto subexpr_it = e.depth_begin(), subexpr_end = e.depth_end();
309 subexpr_it != subexpr_end;
310 ++subexpr_it)
311 {
312 if(subexpr_it->id() == ID_dereference)
313 {
314 if(!first)
315 out << ", ";
316 first = true;
317 format_rec(out, to_dereference_expr(*subexpr_it).pointer());
318 }
319 }
320 });
321 out << "}";
322 }
323
324 out << '\n';
325 instruction.output(out);
326 out << '\n';
327 }
