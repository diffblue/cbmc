32inline if_exprt lower_gcc_conditional_expression(const side_effect_exprt &expr)
33{
34 PRECONDITION(expr.get_statement() == ID_gcc_conditional_expression);
35 const binary_exprt &binary = to_binary_expr(expr);
36 if_exprt if_expr{
37 typecast_exprt::conditional_cast(binary.op0(), bool_typet{}),
38 binary.op0(),
39 binary.op1(),
40 expr.type()};
41 if_expr.add_source_location() = expr.source_location();

43}

52inline void lower_gcc_conditional_expressions(exprt &expr)
53{
54 for(auto &op : expr.operands())
55 lower_gcc_conditional_expressions(op);
56
57 if(
58 expr.id() == ID_side_effect &&

61 expr.operands().size() == 2)

63 if_exprt if_expr =
