80template <typename T>
81struct numeric_castt<T,
82 typename std::enable_if<std::is_integral<T>::value>::type>
83{
84private:
85 // Unchecked conversion from mp_integer when T is signed
86 template <typename U = T,
87 typename std::enable_if<std::is_signed<U>::value, int>::type = 0>
88 static auto get_val(const mp_integer &mpi) -> decltype(mpi.to_long())
89 {
90 return mpi.to_long();
91 }
92
93 // Unchecked conversion from mp_integer when T is unsigned
94 template <typename U = T,
95 typename std::enable_if<!std::is_signed<U>::value, int>::type = 0>
96 static auto get_val(const mp_integer &mpi) -> decltype(mpi.to_ulong())
97 {
98 return mpi.to_ulong();
99 }
100
101public:
102 // Conversion from mp_integer to integral type T
103 std::optional<T> operator()(const mp_integer &mpi) const
104 {
105 static_assert(
106 std::numeric_limits<T>::max() <=
107 std::numeric_limits<decltype(get_val(mpi))>::max() &&
108 std::numeric_limits<T>::min() >=
109 std::numeric_limits<decltype(get_val(mpi))>::min(),
110 "Numeric cast only works for types smaller than long long");
111
112 if(
113 mpi <= std::numeric_limits<T>::max() &&
114 mpi >= std::numeric_limits<T>::min())
115 // to_long converts to long long which is the largest signed numeric type
116 return static_cast<T>(get_val(mpi));
117 else
118 return std::nullopt;
119 }
120
121 // Conversion from expression
122 std::optional<T> operator()(const exprt &expr) const
123 {
124 if(expr.is_constant())
125 return numeric_castt<T>{}(to_constant_expr(expr));
126 else
127 return std::nullopt;
128 }
129
130 // Conversion from expression
131 std::optional<T> operator()(const constant_exprt &expr) const
132 {
133 if(auto mpi_opt = numeric_castt<mp_integer>{}(expr))
134 return numeric_castt<T>{}(*mpi_opt);
135 else
136 return std::nullopt;
137 }
138};
