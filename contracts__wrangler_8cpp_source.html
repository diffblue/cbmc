33contracts_wranglert::contracts_wranglert(
34 goto_modelt &goto_model,
35 const std::string &file_name,
36 message_handlert &message_handler)
37 : ns(goto_model.symbol_table),
38 goto_model(goto_model),
39 symbol_table(goto_model.symbol_table),
40 goto_functions(goto_model.goto_functions),
41 message_handler(message_handler)
42{
43 jsont configuration;
44
45 if(parse_json(file_name, message_handler, configuration))
46 throw deserialization_exceptiont(file_name + " is not a valid JSON file");
47
48 configure_functions(configuration);
49
50 // Mangle loop contracts function by function.
51 // We match function with a regex. There should one and only one function
52 // with name matched by the regex.
53 for(const auto &function_entry : this->functions)
54 {
55 bool function_found = false;
56
57 for(const auto &function : goto_model.goto_functions.function_map)
58 {
59 // We find a matched function.
60 if(std::regex_match(function.first.c_str(), function_entry.first))
61 {
62 if(function_found)
63 throw deserialization_exceptiont(
64 "function regex " + function_entry.second.regex_str +
65 " matches more than one function");
66
67 function_found = true;
68
69 // Mangle all loop contracts in the function.
70 for(const auto &loop_entry : function_entry.second.loop_contracts)
71 {
72 mangle(loop_entry, function.first);
73 }
74 }
75 }
76
77 if(!function_found)
78 throw deserialization_exceptiont(
79 "function regex " + function_entry.second.regex_str +
80 " matches no function");
81 }
82}

84std::optional<irep_idt>
85contracts_wranglert::add_builtin_pointer_function_symbol(
86 std::string function_name,
87 const std::size_t num_of_args)
88{
89 // Already exist.
90 if(symbol_table.has_symbol(CPROVER_PREFIX + function_name))
91 return {};
92
93 code_typet::parameterst parameters;
94 for(unsigned i = 0; i < num_of_args; i++)
95 {
96 parameters.emplace_back(pointer_type(void_type()));
97 }
98 symbolt fun_symbol(
99 CPROVER_PREFIX + function_name,
100 code_typet(parameters, empty_typet()),
101 ID_C);
102 symbol_table.add(fun_symbol);
103 return fun_symbol.name;
104}

108static std::string
109describe_missing_loop(goto_functiont &function, const unsigned loop_number)
110{
111 for(auto it = function.body.instructions.begin();
112 it != function.body.instructions.end();
113 ++it)
114 {
115 if(!it->is_backwards_goto() || it->loop_number != loop_number)
116 continue;
117
118 if(it->get_target() == it)
119 {
120 return "consists of a single instruction that jumps to itself, which "
121 "loop contracts do not support";
122 }
123
124 // A loop head with several back edges forms a single natural loop, which
125 // is numbered by its last back edge.
126 for(auto other = std::next(it); other != function.body.instructions.end();
127 ++other)
128 {
129 if(other->is_backwards_goto() && other->get_target() == it->get_target())
130 {
131 const auto loop = find_loop_head_and_end(other->loop_number, function);
132 if(loop.has_value())
133 {
134 return "is not the last back edge to its loop head; contracts for "
135 "this loop must be given for loop " +
136 std::to_string(loop->second->loop_number);
137 }
138 }
139 }
140
141 return "is not a natural loop";
142 }
143
144 return "does not exist";
145}

147void contracts_wranglert::mangle(
148 const loop_contracts_clauset &loop_contracts,
149 const irep_idt &function_id)
150{
151 messaget log(message_handler);
152
153 const std::string loop_description =
154 "loop " + std::to_string(loop_contracts.loop_number) + " of function " +
155 id2string(function_id);
156
157 auto function_it = goto_model.goto_functions.function_map.find(function_id);
158 PRECONDITION(function_it != goto_model.goto_functions.function_map.end());
159 const auto loop =
160 find_loop_head_and_end(loop_contracts.loop_number, function_it->second);
161 if(!loop.has_value())
162 {
163 throw invalid_input_exceptiont{
164 loop_description + ", as given in the loop contracts file, " +
165 describe_missing_loop(function_it->second, loop_contracts.loop_number)};
166 }
167 const goto_programt::targett loop_end = loop->second;
168
169 // Loop contracts mangling consists of four steps.
170 //
171 // 1. Construct a fake function with a fake loop that contains the loop
172 // contracts for parsing the contracts.
173 // 2. Parse the fake function and extract the contracts as exprt.
174 // 3. Replace symbols in the extracted exprt with the symbols in the
175 // symbol table of the goto model according to the symbol map provided by
176 // the user.
177 // 4. Typecheck all contracts exprt.
178 // 5. Annotate all contracts exprt to the corresponding loop.
179
180 // Construct a fake function to parse.
181 // void function_id()
182 // {
183 // while(1==1)
184 // __CPROVER_assigns(assigns_str) // optional
185 // __CPROVER_loop_invariant(invariants_str)
186 // __CPROVER_decreases(decreases_str) // optional
187 // }
188 std::ostringstream pr;
189 pr << "void _fn() { while(1==1)";
190 if(!loop_contracts.assigns.empty())
191 {
192 pr << CPROVER_PREFIX << "assigns(" << loop_contracts.assigns << ") ";
193 }
194 pr << CPROVER_PREFIX << "loop_invariant(" << loop_contracts.invariants
195 << ") ";
196 if(!loop_contracts.decreases.empty())
197 {
198 pr << CPROVER_PREFIX << "decreases(" << loop_contracts.decreases << ") ";
199 }
200 pr << " {}}" << std::endl;
201
202 log.debug() << "Constructing fake function:\n" << pr.str() << log.eom;
203
204 // Parse the fake function.
205 std::istringstream is(pr.str());
206 ansi_c_parsert ansi_c_parser{message_handler};
207 ansi_c_parser.set_line_no(0);
208 ansi_c_parser.set_file("<predicate>");
209 ansi_c_parser.in = &is;
210 ansi_c_parser.for_has_scope = config.ansi_c.for_has_scope;
211 ansi_c_parser.ts_18661_3_Floatn_types = config.ansi_c.ts_18661_3_Floatn_types;
212 ansi_c_parser.__float128_is_keyword = config.ansi_c.__float128_is_keyword;
213 ansi_c_parser.float16_type = config.ansi_c.float16_type;
214 ansi_c_parser.bf16_type = config.ansi_c.bf16_type;
215 ansi_c_parser.fp16_type = config.ansi_c.fp16_type;
216 ansi_c_parser.cpp98 = false; // it's not C++
217 ansi_c_parser.cpp11 = false; // it's not C++
218 ansi_c_parser.mode = config.ansi_c.mode;
219 ansi_c_scanner_init(ansi_c_parser);
220
221 // The clauses are user input: a syntax error leaves the parse tree empty,
222 // which must be reported rather than dereferenced.
223 if(ansi_c_parser.parse() || ansi_c_parser.parse_tree.items.empty())
224 {
225 throw invalid_input_exceptiont{
226 "failed to parse the loop contracts for " + loop_description +
227 " given in the loop contracts file"};
228 }
229
230 // The loop of the fake function, which carries the clauses.
231 codet &fake_loop = static_cast<codet &>(
232 ansi_c_parser.parse_tree.items.front().declarator().value().operands()[0]);
233 // As in the C front-end, decide from the parse tree which clauses are given.
234 // Note that add() would create an empty (non-nil) clause.
235 const bool has_assigns = fake_loop.find(ID_C_spec_assigns).is_not_nil();
236 const bool has_decreases = fake_loop.find(ID_C_spec_decreases).is_not_nil();
237
238 // Replace symbols in the clauses with the symbols in the symbol table.
239 log.debug() << "Start to replace symbols" << log.eom;
240 loop_contracts.replace_symbol(
241 static_cast<exprt &>(fake_loop.add(ID_C_spec_loop_invariant)));
242 if(has_assigns)
243 loop_contracts.replace_symbol(
244 static_cast<exprt &>(fake_loop.add(ID_C_spec_assigns)));
245 if(has_decreases)
246 loop_contracts.replace_symbol(
247 static_cast<exprt &>(fake_loop.add(ID_C_spec_decreases)));
248
249 // Add builtin functions that might be used in contracts to the symbol table,
250 // and remove them again when leaving this function, also via an exception.
251 struct remove_symbols_on_exitt
252 {
253 symbol_tablet &symbol_table;
254 std::vector<irep_idt> names;
255 ~remove_symbols_on_exitt()
256 {
257 for(const auto &name : names)
258 symbol_table.remove(name);
259 }
260 } added_symbols{symbol_table, {}};
261 for(const auto &builtin : std::vector<std::pair<std::string, std::size_t>>{
262 {"loop_entry", 1},
263 {"same_object", 2},
264 {"OBJECT_SIZE", 1},
265 {"POINTER_OBJECT", 1},
266 {"POINTER_OFFSET", 1}})
267 {
268 if(
269 const auto added =
270 add_builtin_pointer_function_symbol(builtin.first, builtin.second))
271 {
272 added_symbols.names.push_back(*added);
273 }
274 }
275
276 // Typecheck the clauses as the C front-end does for loop contracts given in
277 // the source code: this converts the invariant to Boolean and the
278 // components of the decreases clause to arithmetic types, and rejects
279 // __CPROVER_old in the invariant. The type checker reports errors via the
280 // message handler and then throws.
281 ansi_c_typecheckt ansi_c_typecheck(
282 ansi_c_parser.parse_tree,
283 symbol_table,
284 symbol_table.lookup_ref(function_id).module.c_str(),
285 log.get_message_handler());
286
287 const auto typecheck_error = [&loop_description]()
288 {
289 return invalid_input_exceptiont{
290 "failed to typecheck the loop contracts for " + loop_description +
291 " given in the loop contracts file"};
292 };
293
294 // The same exceptions as typecheckt::typecheck_main handles.
295 try
296 {
297 log.debug() << "Start to typecheck the clauses." << log.eom;
298 ansi_c_typecheck.typecheck_spec_loop_invariant(fake_loop);
299 if(has_assigns)
300 {
301 ansi_c_typecheck.typecheck_spec_assigns(
302 static_cast<unary_exprt &>(fake_loop.add(ID_C_spec_assigns))
303 .op()
304 .operands());
305 }
306 ansi_c_typecheck.typecheck_spec_decreases(fake_loop);
307 }
308 catch(int)
309 {
310 throw typecheck_error();
311 }
312 catch(const char *e)
313 {
314 log.error() << e << messaget::eom;
315 throw typecheck_error();
316 }
317 catch(const std::string &e)
318 {
319 log.error() << e << messaget::eom;
320 throw typecheck_error();
321 }
322 catch(const typecheckt::errort &e)
323 {
324 if(!e.what().empty())
325 {
326 log.error().source_location = e.source_location();
327 log.error() << e.what() << messaget::eom;
328 }
329 throw typecheck_error();
330 }
331 catch(const invalid_source_file_exceptiont &e)
332 {
333 log.error().source_location = e.get_source_location();
334 log.error() << e.get_reason() << messaget::eom;
335 throw typecheck_error();
336 }
337
338 // Annotate the invariant. The clause may consist of several invariants, as
339 // in "i <= 8) __CPROVER_loop_invariant(i >= 0", all of which must hold.
340 const exprt::operandst &invariants =
341 static_cast<const exprt &>(fake_loop.find(ID_C_spec_loop_invariant))
342 .operands();
343 const exprt inv_expr = simplify_expr(conjunction(invariants), ns);
344 log.debug() << "Extracted loop invariants: " << inv_expr.pretty() << "\n"
345 << log.eom;
346 loop_end->condition_nonconst().add(ID_C_spec_loop_invariant) =
347 and_exprt{inv_expr, true_exprt{}};
348
349 // Annotate the assigns clause.
350 if(has_assigns)
351 {
352 loop_end->condition_nonconst().add(ID_C_spec_assigns) =
353 static_cast<const exprt &>(fake_loop.find(ID_C_spec_assigns))
354 .operands()[0];
355 }
356
357 // Annotate the decreases clause.
358 if(has_decreases)
359 {
360 exprt decreases{ID_target_list};
361 decreases.operands() =
362 static_cast<const exprt &>(fake_loop.find(ID_C_spec_decreases))
363 .operands();
364 loop_end->condition_nonconst().add(ID_C_spec_decreases) = decreases;
365 }
366
367 log.debug() << "Mangling finished." << log.eom;
368}

370void contracts_wranglert::configure_functions(const jsont &config)
371{
372 auto functions = config["functions"];
373
374 if(functions.is_null())
375 return;
376
377 if(!functions.is_array())
378 throw deserialization_exceptiont("functions entry must be sequence");
379
380 for(const auto &function : to_json_array(functions))
381 {
382 if(!function.is_object())
383 throw deserialization_exceptiont("function entry must be object");
384
385 for(const auto &function_entry : to_json_object(function))
386 {
387 const auto function_name = function_entry.first;
388 const auto &items = function_entry.second;
389
390 if(!items.is_array())
391 throw deserialization_exceptiont("loops entry must be sequence");
392
393 this->functions.emplace_back(function_name, functiont{});
394 functiont &function_config = this->functions.back().second;
395 function_config.regex_str = function_name;
396
397 for(const auto &function_item : to_json_array(items))
398 {
399 if(!function_item.is_object())
400 throw deserialization_exceptiont("loop entry must be object");
401
402 std::string loop_id = "";
403 std::string invariants_str = "";
404 std::string assigns_str = "";
405 std::string decreases_str = "";
406 unchecked_replace_symbolt replace_symbol;
407
408 for(const auto &loop_item : to_json_object(function_item))
409 {
410 if(!loop_item.second.is_string())
411 throw deserialization_exceptiont("loop item must be string");
412
413 if(loop_item.first == "loop_id")
414 {
415 loop_id = loop_item.second.value;
416 continue;
417 }
418
419 if(loop_item.first == "assigns")
420 {
421 assigns_str = loop_item.second.value;
422 continue;
423 }
424
425 if(loop_item.first == "decreases")
426 {
427 decreases_str = loop_item.second.value;
428 continue;
429 }
430
431 if(loop_item.first == "invariants")
432 {
433 invariants_str = loop_item.second.value;
434 continue;
435 }
436
437 if(loop_item.first == "symbol_map")
438 {
439 std::string symbol_map_str = loop_item.second.value;
440
441 // Remove all space in symbol_map_str
442 symbol_map_str.erase(
443 std::remove_if(
444 symbol_map_str.begin(), symbol_map_str.end(), isspace),
445 symbol_map_str.end());
446
447 for(const auto &symbol_map_entry :
448 split_string(symbol_map_str, ';'))
449 {
450 const auto symbol_map_pair = split_string(symbol_map_entry, ',');
451 const auto &symbol = symbol_table.lookup(symbol_map_pair.back());
452 if(symbol == nullptr)
453 throw deserialization_exceptiont(
454 "symbol with id \"" + symbol_map_pair.front() +
455 "\" does not exist in the symbol table");
456 // Create a dummy symbol. The type will be discarded when inserted
457 // into replace_symbol.
458 const symbol_exprt old_expr(
459 symbol_map_pair.front(), bool_typet());
460 replace_symbol.insert(old_expr, symbol->symbol_expr());
461 }
462
463 continue;
464 }
465
466 throw deserialization_exceptiont("unexpected loop item");
467 }
468
469 if(loop_id.empty())
470 {
471 throw deserialization_exceptiont(
472 "loop entry must have one identifier");
473 }
474
475 // Surrounding white space is permitted, as in the symbol map.
476 const auto first = loop_id.find_first_not_of(" \t\n\r");
477 const auto last = loop_id.find_last_not_of(" \t\n\r");
478 const std::optional<unsigned> loop_number =
479 first == std::string::npos
480 ? std::optional<unsigned>{}
481 : string2optional_unsigned(loop_id.substr(first, last - first + 1));
482 if(!loop_number.has_value())
483 {
484 throw deserialization_exceptiont(
485 "loop identifier '" + loop_id + "' of function " + function_name +
486 " given in the loop contracts file is not a loop number");
487 }
488
489 if(invariants_str.empty())
490 {
491 throw deserialization_exceptiont(
492 "loop entry must have one invariant string");
493 }
494
495 function_config.loop_contracts.emplace_back(
496 *loop_number,
497 invariants_str,
498 assigns_str,
499 decreases_str,
500 replace_symbol);
501 }
502 }
503 }
