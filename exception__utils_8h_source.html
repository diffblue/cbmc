27class cprover_exception_baset
28{
29public:
33 virtual std::string what() const;
34 virtual ~cprover_exception_baset() = default;
35
36 cprover_exception_baset(cprover_exception_baset &&) = default;
37
38 // std::ostringstream does not have a copy constructor
39 cprover_exception_baset(const cprover_exception_baset &src)
40 {
41 reason_stream << src.reason();
42 }
43
44 // std::ostringstream is not copy-assignable, and exceptions need not be
45 // assignable
46 cprover_exception_baset &operator=(const cprover_exception_baset &) = delete;
47
48 std::string reason() const
49 {
50 return reason_stream.str();
51 }
52
53protected:
57 cprover_exception_baset() = default;
58
59 explicit cprover_exception_baset(std::string reason)
60 : reason_stream{std::move(reason), std::ios_base::out | std::ios_base::ate}
61 {
62 }
63
67 std::ostringstream reason_stream;
68
69 // to provide access to the reason_stream field
70 template <typename E, typename T>
71 friend std::enable_if_t<
72 std::is_base_of<cprover_exception_baset, std::decay_t<E>>::value,
73 E>
74 operator<<(E &&, const T &);
75};
76
84template <typename E, typename T>
85std::
86 enable_if_t<
87 std::is_base_of<cprover_exception_baset, std::decay_t<E>>::value,
88 E>
89 operator<<(E &&e, const T &message)
90{
91 e.reason_stream << message;
92 return std::forward<E>(e);
93}
