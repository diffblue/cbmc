27class cprover_exception_baset
28{
29public:
33 virtual std::string what() const;
34 virtual ~cprover_exception_baset() = default;
35
36 cprover_exception_baset() = default;
37
38 cprover_exception_baset(cprover_exception_baset &&) = default;
39
40 // std::ostringstream does not have a copy constructor
41 cprover_exception_baset(const cprover_exception_baset &src)
42 {
43 _reason << src.reason();
44 }
45
46 std::string reason() const
47 {
48 return _reason.str();
49 }
50
51protected:
55 explicit cprover_exception_baset(std::string reason)
56 {
57 // ostringstream does not have an appropriate constructor
58 _reason << reason;
59 }
60
64 std::ostringstream _reason;
65
66 // to provide access to the _reason field
67 template <typename E, typename T>
68 friend std::enable_if_t<
69 std::is_base_of<cprover_exception_baset, std::decay_t<E>>::value,
70 E &&>
71 operator<<(E &&, const T &);
72};
73
78template <typename E, typename T>
79std::enable_if_t<
80 std::is_base_of<cprover_exception_baset, std::decay_t<E>>::value,
81 E &&>
82operator<<(E &&e, const T &message)
83{
84 e._reason << message;
85 return std::forward<E>(e);
86}
