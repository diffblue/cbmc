50static bool
51is_region_instruction(const goto_programt::instructiont &instruction)
52{
53 if(instruction.is_assign())
54 {
55 const exprt &lhs = instruction.assign_lhs();
56 if(has_subexpr(lhs, ID_dereference) || has_subexpr(lhs, ID_index))
57 return false;
58 return !has_subexpr(
59 instruction.assign_rhs(),
60 [](const exprt &e)
61 {
62 return e.id() == ID_side_effect &&
63 to_side_effect_expr(e).get_statement() != ID_nondet;
64 });
65 }
66
67 return instruction.is_skip() || instruction.is_decl() ||
68 instruction.is_dead();
69}

106std::size_t if_conversion(
107 goto_programt &goto_program,
108 symbol_table_baset &symbol_table,
109 const irep_idt &mode,
110 const namespacet &ns)
111{
112 // Map each instruction to the set of instructions that jump to it, so that
113 // we can ensure that the branch we remove is the only way into the region we
114 // linearise (and that the branch itself is not a jump target). Built once;
115 // it is treated conservatively (a stale entry only prevents a later
116 // conversion, it never enables an unsound one).
117 std::map<
118 const goto_programt::instructiont *,
119 std::set<const goto_programt::instructiont *>>
120 incoming;
121 for(auto it = goto_program.instructions.begin();
122 it != goto_program.instructions.end();
123 ++it)
124 {
125 for(const auto &t : it->targets)
126 incoming[&*t].insert(&*it);
127 }
128
129 const auto externally_entered =
130 [&incoming](
131 goto_programt::targett p,
132 const goto_programt::instructiont *branch,
133 const goto_programt::instructiont *structural)
134 {
135 const auto m = incoming.find(&*p);
136 if(m == incoming.end())
137 return false;
138 for(const auto *src : m->second)
139 if(src != branch && src != structural)
140 return true;
141 return false;
142 };
143
144 std::size_t converted = 0;
145 std::size_t guard_counter = 0;
146
147 for(auto it = goto_program.instructions.begin();
148 it != goto_program.instructions.end();
149 ++it)
150 {
151 // candidate: a conditional GOTO with a single, forward target
152 if(!it->is_goto() || it->targets.size() != 1)
153 continue;
154
155 const exprt guard = it->condition();
156 if(guard.is_true() || guard.is_false())
157 continue;
158
159 // the branch itself must not be a jump target, so that the guard snapshot
160 // we insert just before it is reached on every entry
161 if(incoming.find(&*it) != incoming.end())
162 continue;
163
164 const goto_programt::targett target = it->get_target();
165
166 // the then-region is [then_begin, target); require target to be reachable
167 // forwards (otherwise this is a back-edge / loop)
168 const goto_programt::targett then_begin = std::next(it);
169 std::vector<goto_programt::targett> region1;
170 bool forward = false;
171 for(auto p = then_begin; p != goto_program.instructions.end(); ++p)
172 {
173 if(p == target)
174 {
175 forward = true;
176 break;
177 }
178 region1.push_back(p);
179 }
180 if(!forward)
181 continue;
182
183 // if-else shape: the then-region ends in an unconditional forward GOTO
184 // whose target (the join) lies beyond the else-entry `target`
185 goto_programt::targett structural_goto = goto_program.instructions.end();
186 std::vector<goto_programt::targett> region2;
187 bool is_if_else = false;
188 if(
189 !region1.empty() && region1.back()->is_goto() &&
190 region1.back()->condition().is_true() &&
191 region1.back()->targets.size() == 1)
192 {
193 const goto_programt::targett join = region1.back()->get_target();
194 bool join_forward = false;
195 for(auto p = target; p != goto_program.instructions.end(); ++p)
196 {
197 if(p == join)
198 {
199 join_forward = true;
200 break;
201 }
202 }
203 if(join_forward && join != target)
204 {
205 structural_goto = region1.back();
206 region1.pop_back();
207 is_if_else = true;
208 for(auto p = target; p != join; ++p)
209 region2.push_back(p);
210 }
211 }
212
213 // both regions must consist of linearisable instructions only
214 const auto linearisable =
215 [](const std::vector<goto_programt::targett> &region)
216 {
217 for(const auto &p : region)
218 if(!is_region_instruction(*p))
219 return false;
220 return true;
221 };
222 if(!linearisable(region1) || !linearisable(region2))
223 continue;
224
225 // at least one region must contain an assignment, otherwise there is
226 // nothing worth linearising
227 const auto has_assignment =
228 [](const std::vector<goto_programt::targett> &region)
229 {
230 for(const auto &p : region)
231 if(p->is_assign())
232 return true;
233 return false;
234 };
235 if(!has_assignment(region1) && !has_assignment(region2))
236 continue;
237
238 // single-entry: no instruction in the region(s), nor the else-entry, may
239 // be the target of any jump other than the branch (and the structural
240 // GOTO) we are removing
241 const goto_programt::instructiont *const structural_ptr =
242 is_if_else ? &*structural_goto : nullptr;
243 bool external_entry = false;
244 for(const auto &p : region1)
245 external_entry |= externally_entered(p, &*it, structural_ptr);
246 for(const auto &p : region2)
247 external_entry |= externally_entered(p, &*it, structural_ptr);
248 if(is_if_else)
249 external_entry |= externally_entered(target, &*it, structural_ptr);
250 if(external_entry)
251 continue;
252
253 // Snapshotting the guard into a fresh variable is necessary when the guard
254 // reads a variable that the region writes or kills: otherwise re-evaluating
255 // the guard at each converted assignment could observe a clobbered value,
256 // or read a now-DEAD condition temporary. When the guard does not depend on
257 // anything the region touches we can use the original condition directly,
258 // which keeps the conditional assignments analysable downstream (e.g. it
259 // preserves `p != null` guards for local safe pointer analysis).
260 // A guard that reads memory via a dereference may observe any write of
261 // the region (through aliasing), so it is always snapshotted.
262 const find_symbols_sett guard_symbols = find_symbol_identifiers(guard);
263 bool guard_depends_on_region = has_subexpr(guard, ID_dereference);
264 const auto note_region_writes = [&guard_symbols, &guard_depends_on_region](
265 const goto_programt::targett &p)
266 {
267 if(p->is_assign())
268 {
269 for(const irep_idt &id : find_symbol_identifiers(p->assign_lhs()))
270 if(guard_symbols.count(id) != 0)
271 guard_depends_on_region = true;
272 }
273 else if(p->is_decl())
274 {
275 if(guard_symbols.count(p->decl_symbol().get_identifier()) != 0)
276 guard_depends_on_region = true;
277 }
278 else if(p->is_dead())
279 {
280 if(guard_symbols.count(p->dead_symbol().get_identifier()) != 0)
281 guard_depends_on_region = true;
282 }
283 };
284 for(const auto &p : region1)
285 note_region_writes(p);
286 for(const auto &p : region2)
287 note_region_writes(p);
288
289 // For an if/else, the else-region is linearised after the then-region, so
290 // converting each region independently into a self-referential conditional
291 // assignment `x = c ? e : x` makes an else-assignment that reads a variable
292 // the then-region wrote nest (after SSA) into `x = !c ? b : (c ? a : x)`.
293 // That defeats constant propagation of `x`: a variable that was concrete on
294 // each original branch becomes a symbolic value, and in --paths mode that
295 // can drive downstream branches/loops into an explosion of paths (see
296 // regression/cbmc/return4). When both regions are a single assignment to
297 // the same variable we instead emit one combined ternary `x = c ? a : b`,
298 // which simplify can fold (e.g. `c ? 1 : 1` to `1`); messier overlaps are
299 // left unconverted. Regions that do not interfere keep the per-region form.
300 goto_programt::targett combine_then = goto_program.instructions.end();
301 goto_programt::targett combine_else = goto_program.instructions.end();
302 bool combine = false;
303 if(is_if_else)
304 {
305 find_symbols_sett region1_writes;
306 for(const auto &p : region1)
307 {
308 if(p->is_assign())
309 {
310 for(const irep_idt &id : find_symbol_identifiers(p->assign_lhs()))
311 region1_writes.insert(id);
312 }
313 else if(p->is_decl())
314 region1_writes.insert(p->decl_symbol().get_identifier());
315 }
316
317 bool region2_uses_region1_write = false;
318 std::vector<goto_programt::targett> region1_assignments;
319 std::vector<goto_programt::targett> region2_assignments;
320 for(const auto &p : region1)
321 if(p->is_assign())
322 region1_assignments.push_back(p);
323 for(const auto &p : region2)
324 {
325 if(p->is_assign())
326 {
327 region2_assignments.push_back(p);
328 for(const irep_idt &id : find_symbol_identifiers(p->assign_lhs()))
329 if(region1_writes.count(id) != 0)
330 region2_uses_region1_write = true;
331 for(const irep_idt &id : find_symbol_identifiers(p->assign_rhs()))
332 if(region1_writes.count(id) != 0)
333 region2_uses_region1_write = true;
334 }
335 else if(
336 p->is_decl() &&
337 region1_writes.count(p->decl_symbol().get_identifier()) != 0)
338 region2_uses_region1_write = true;
339 else if(
340 p->is_dead() &&
341 region1_writes.count(p->dead_symbol().get_identifier()) != 0)
342 region2_uses_region1_write = true;
343 }
344
345 if(region2_uses_region1_write)
346 {
347 if(
348 region1_assignments.size() == 1 && region2_assignments.size() == 1 &&
349 region1_assignments.front()->assign_lhs() ==
350 region2_assignments.front()->assign_lhs())
351 {
352 combine = true;
353 combine_then = region1_assignments.front();
354 combine_else = region2_assignments.front();
355 }
356 else
357 continue; // leave the branch in place rather than create a chain
358 }
359 }
360
361 // the then-region executes when the GOTO is *not* taken, the else-region
362 // when it is taken
363 exprt then_condition;
364 exprt else_condition;
365 if(guard_depends_on_region)
366 {
367 const symbol_exprt snapshot = fresh_guard_symbol(
368 symbol_table, mode, it->source_location(), guard_counter);
369 goto_program.insert_before(
370 it,
371 goto_programt::make_assignment(snapshot, guard, it->source_location()));
372 then_condition = boolean_negate(snapshot);
373 else_condition = snapshot;
374 }
375 else
376 {
377 then_condition = boolean_negate(guard);
378 else_condition = guard;
379 }
380 simplify(then_condition, ns);
381
382 if(combine)
383 {
384 // x = c ? then_rhs : else_rhs, replacing the matching assignment pair;
385 // the else-assignment becomes a skip
386 exprt combined = if_exprt{
387 then_condition, combine_then->assign_rhs(), combine_else->assign_rhs()};
388 simplify(combined, ns);
389 combine_then->assign_rhs_nonconst() = std::move(combined);
390 *combine_else = goto_programt::make_skip(combine_else->source_location());
391 }
392 else
393 {
394 for(const auto &p : region1)
395 {
396 if(!p->is_assign())
397 continue;
398 const exprt lhs = p->assign_lhs();
399 const exprt rhs = p->assign_rhs();
400 p->assign_rhs_nonconst() = if_exprt{then_condition, rhs, lhs};
401 }
402 for(const auto &p : region2)
403 {
404 if(!p->is_assign())
405 continue;
406 const exprt lhs = p->assign_lhs();
407 const exprt rhs = p->assign_rhs();
408 p->assign_rhs_nonconst() = if_exprt{else_condition, rhs, lhs};
409 }
410 }
411
412 // drop the branch(es)
413 *it = goto_programt::make_skip(it->source_location());
414 if(is_if_else)
415 *structural_goto =
416 goto_programt::make_skip(structural_goto->source_location());
417
418 ++converted;
419 }
420
421 return converted;
422}

424std::size_t
425if_conversion(goto_modelt &goto_model, message_handlert &message_handler)
426{
427 // The conversion makes conditional writes unconditional (`x = c ? e : x`),
428 // which another thread could observe: a write of that thread between the
429 // read and the write of `x` would be lost. Hence do not convert anything
430 // in concurrent programs.
431 for(const auto &gf : goto_model.goto_functions.function_map)
432 {
433 for(const auto &instruction : gf.second.body.instructions)
434 {
435 if(instruction.is_start_thread())
436 return 0;
437 }
438 }
439
440 const namespacet ns(goto_model.symbol_table);
441 std::size_t converted = 0;
442
443 for(auto &gf : goto_model.goto_functions.function_map)
444 {
445 const symbolt *fn = goto_model.symbol_table.lookup(gf.first);
446 const irep_idt mode = fn != nullptr ? fn->mode : ID_C;
447 converted +=
448 if_conversion(gf.second.body, goto_model.symbol_table, mode, ns);
449 }
450
451 if(converted != 0)
452 {
453 remove_skip(goto_model);
454 goto_model.goto_functions.update();
455
456 messaget log(message_handler);
457 log.statistics() << "if-conversion replaced " << converted
458 << " branch(es) with conditional assignments"
459 << messaget::eom;
460 }
461
462 return converted;
463}
